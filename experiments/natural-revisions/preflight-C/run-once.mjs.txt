import {execFileSync} from 'node:child_process';
import {resolve} from 'node:path';
import {loadFrozen} from './prepare.mjs';
import {runOnce} from '../decision-ab/runner.mjs';
if(process.argv[2]!=='C'||process.argv[3]!=='--live'||process.argv.length!==4)throw Error('Use explicit C --live');
const root=process.env.PERSPECTA_NATURAL_C_REPORT_ROOT;if(!root||resolve(root)!==root)throw Error('Explicit absolute private report root required');
const frozen=loadFrozen();frozen.metadata.codeCommit=execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8',env:{PATH:'/usr/bin:/bin'}}).trim();
// Reuse the frozen one-attempt runner and protocol A: the response contract and graph are unchanged.
// The request and source version are C, never an A/B rerun. Keep original upstream report.variant=A as protocol metadata.
try{
 const r=await runOnce({variant:'A',request:frozen.request,metadata:frozen.metadata,reportDir:root+'/C',apiKey:process.env.LLM_API_KEY});
 console.log(JSON.stringify({studyVariant:'C',protocolVariant:r.variant,attempts:r.attempts,upstreamStatus:r.upstreamStatus,outcome:r.outcome,usage:r.usage,durationMs:r.durationMs}));process.exitCode=r.outcome==='VALID'?0:1;
}catch{console.log(JSON.stringify({studyVariant:'C',outcome:'C_RUN_FAILED_NO_RETRY'}));process.exitCode=1;}
